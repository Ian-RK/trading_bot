import warnings; warnings.filterwarnings("ignore")
import backtesting_engine as eng
from strategies.price_action_strategy import engineer_features
from utils.ai_compare import compare, buy_and_hold, metrics
for n in ["SPY_NEUTRAL_2018","SPY_BEAR_2022","SPY_BULL_2025","QQQ_TECH_2025","SPY_historical"]:
    df = engineer_features(f"data/{n}.csv").reset_index(drop=True)
    first = int(df["sma_200"].first_valid_index())
    p = compare(df, {})["table"].loc["price_only"]
    bh = metrics(buy_and_hold(df.iloc[first:].reset_index(drop=True), eng.INITIAL_CAPITAL,
                              eng.COMMISSION_PER_SHARE, eng.SLIPPAGE_PCT), eng.INITIAL_CAPITAL)
    print(n, f"rows={len(df)} first_signal_row={first} evaluable={len(df)-first} | strategy {p.total_return_pct:.2f}% trades={int(p.trades)} maxDD={p.max_drawdown_pct:.2f}% | buy&hold same window {bh['total_return_pct']:.2f}% maxDD={bh['max_drawdown_pct']:.2f}%")
